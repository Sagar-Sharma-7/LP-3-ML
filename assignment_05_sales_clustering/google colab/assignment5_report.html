In [6]:
wcss = []

for k in range(1, 9):
    model = KMeans(
        n_clusters=k,
        random_state=42,
        n_init=10
    )
    model.fit(X_scaled)
    wcss.append(model.inertia_)

plt.figure(figsize=(7, 4))
plt.plot(range(1, 9), wcss, marker="o")
plt.title("Elbow Method")
plt.xlabel("Number of Clusters (K)")
plt.ylabel("WCSS")
plt.xticks(range(1, 9))
plt.grid(True)
plt.tight_layout()
plt.savefig("elbow_method.png", dpi=200)
plt.show()
